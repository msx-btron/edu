\chapter{Analytical Waveforms, Spectral Synthesis, and Digital Signal Processing}
\label{ch:waveforms-dsp}

From the algebraic pitch frequencies $f(n)$ of Chapter~\ref{ch:group-theory}, a Virtual DSP constructs time-domain waveforms $w(\varphi)$, shapes their harmonic spectra through linear time-invariant (LTI) filters and non-linear wave-shapers, and embeds them in an acoustic space via delay networks. This chapter rigorously formalizes all thirteen DSP primitives of B-Sound, proving their spectral structure, stability, and dynamic bounds.

\section{Continuous Phase Integration and Canonical Waveforms}

\begin{definition}[Continuous Phase Accumulator]
Given an instantaneous frequency trajectory $f : \mathbb{R}_{\ge 0} \to \mathbb{R}_{>0}$ that is piecewise continuous, the \emph{continuous phase accumulator} is the integral
\begin{equation}
\label{eq:phase-acc}
\varphi(t) = \left(2\pi \int_0^t f(\tau)\,d\tau\right) \bmod 2\pi,
\end{equation}
discretized at sample rate $f_s$ by the recurrence $\varphi[n] = (\varphi[n-1] + 2\pi f[n]/f_s) \bmod 2\pi$.
\end{definition}

\begin{definition}[Canonical B-Sound Periodic Waveform Basis]
The four periodic oscillator primitives of B-Sound are the $2\pi$-periodic functions $w : \mathbb{T} \to [-1, 1]$ defined by:
\begin{align}
w_{\mathrm{sin}}(\varphi) &= \sin(\varphi) \qquad \big(\text{sub-bass } f < 95\,\text{Hz}: 0.90\sin\varphi + 0.10\sin 2\varphi\big), \\
w_{\mathrm{tri}}(\varphi) &= \frac{2}{\pi}\arcsin(\sin\varphi) = \frac{8}{\pi^2}\sum_{k=0}^{\infty}\frac{(-1)^k}{(2k+1)^2}\sin\!\big((2k+1)\varphi\big), \\
w_{\mathrm{saw}}(\varphi) &= 0.62 \sum_{k=1}^{4} \frac{(-1)^{k-1}}{k}\sin(k\varphi), \\
w_{\mathrm{pls}}(\varphi) &= 0.78 \tanh(3.4\sin\varphi).
\end{align}
\end{definition}

\begin{theorem}[Phase Continuity, Odd-Harmonic Symmetry, and Alias-Free Band-Limit]
\label{thm:waveforms}
Let $f(t)$ be bounded and piecewise continuous with $0 < f(t) \le f_{\max}$. Then:
\begin{enumerate}[leftmargin=1.5em]
  \item Every composition $w(\varphi(t))$ for $w \in \{w_{\mathrm{sin}}, w_{\mathrm{tri}}, w_{\mathrm{saw}}, w_{\mathrm{pls}}\}$ is globally continuous ($C^0$) in $t$, eliminating click discontinuities during arbitrary pitch steps.
  \item Both $w_{\mathrm{tri}}$ and $w_{\mathrm{pls}}$ satisfy half-wave anti-symmetry $w(\varphi + \pi) = -w(\varphi)$ and odd parity $w(-\varphi) = -w(\varphi)$, so their Fourier expansions contain exclusively odd sine harmonics $b_{2k+1}\sin((2k+1)\varphi)$.
  \item The band-limited sawtooth $w_{\mathrm{saw}}$ is strictly alias-free at sample rate $f_s$ whenever $f_{\max} < f_s / 8$ ($f_{\max} < 6000\,\text{Hz}$ at $f_s = 48000\,\text{Hz}$, covering MIDI notes $n \le 114$).
\end{enumerate}
\end{theorem}

\begin{proof}
(1) Since $f$ is bounded and integrable, $\int_0^t f(\tau)\,d\tau$ is Lipschitz continuous, and every $w$ is continuous on $\mathbb{R}/2\pi\mathbb{Z}$; hence $w(\varphi(t))$ is $C^0$.
(2) For $w_{\mathrm{pls}}$, $\sin(\varphi + \pi) = -\sin\varphi$ and $\tanh(-u) = -\tanh u$, so $w_{\mathrm{pls}}(\varphi+\pi) = -w_{\mathrm{pls}}(\varphi)$ and $w_{\mathrm{pls}}(-\varphi) = -w_{\mathrm{pls}}(\varphi)$; identical identities hold for $\arcsin(\sin\varphi)$. Consequently all cosine coefficients $a_m$ and all even sine coefficients $b_{2k}$ vanish identically.
(3) The highest partial in $w_{\mathrm{saw}}$ is $k = 4$, with frequency $4 f_{\max}$. By the Shannon--Nyquist sampling theorem, aliasing is absent if and only if $4 f_{\max} < f_s / 2$, i.e.\ $f_{\max} < f_s / 8$. At $f_s = 48000\,\text{Hz}$, $f_s / 8 = 6000\,\text{Hz} > f(114) = 5919.91\,\text{Hz}$.
\end{proof}

\section{Frequency Modulation (FM) and Jacobi--Anger Spectral Theory}

\begin{definition}[Two-Operator Frequency Modulation (Equation~12)]
For carrier frequency $f_c > 0$, modulator frequency $f_m = r f_c$, amplitude envelope $A_c(t)$, and modulation index $I(t) \ge 0$, \emph{two-operator FM synthesis} is defined by
\begin{equation}
\label{eq:fm-synth}
y_{\mathrm{FM}}(t) = A_c(t)\sin\!\Big(2\pi f_c t + I(t)\sin(2\pi f_m t)\Big).
\end{equation}
\end{definition}

\begin{theorem}[Jacobi--Anger Spectral Decomposition of FM]
\label{thm:jacobi-anger}
For constant $A_c = 1$ and constant modulation index $I \ge 0$, the FM signal \eqref{eq:fm-synth} admits the exact uniformly convergent Fourier expansion
\begin{equation}
\label{eq:bessel-exp}
\sin\!\big(\omega_c t + I\sin(\omega_m t)\big) = \sum_{m=-\infty}^{\infty} J_m(I)\sin\!\big((\omega_c + m\omega_m)t\big),
\end{equation}
where $J_m(I)$ is the Bessel function of the first kind of order $m$ satisfying $J_{-m}(I) = (-1)^m J_m(I)$ and Parseval energy conservation $\sum_{m=-\infty}^{\infty} J_m(I)^2 = 1$.
\end{theorem}

\begin{proof}
Recall the Jacobi--Anger generating identity $e^{i I \sin\theta} = \sum_{m=-\infty}^{\infty} J_m(I)\,e^{i m\theta}$, which converges uniformly on $\theta \in \mathbb{R}$. Multiplying both sides by $e^{i\omega_c t}$ with $\theta = \omega_m t$ yields
\[
e^{i(\omega_c t + I\sin(\omega_m t))} = \sum_{m=-\infty}^{\infty} J_m(I)\,e^{i(\omega_c + m\omega_m)t}.
\]
Taking the imaginary part of both sides gives \eqref{eq:bessel-exp}. Since $|e^{i I \sin\theta}|^2 = 1$, Parseval's identity on $\mathbb{T}$ yields $\sum_{m=-\infty}^{\infty} |J_m(I)|^2 = 1$, confirming that increasing $I$ redistributes energy from the carrier into sidebands without altering total RMS power.
\end{proof}

\begin{corollary}[Inharmonic Bell Spectrum of \texttt{shimPulseBell}, GM Program 98]
Setting the irrational ratio $r = f_m/f_c = 2.76$ and $I = 1.8$ places significant sideband energy ($|J_m(1.8)| > 0.05$ for $|m| \le 3$) at non-integer multiples $f_c$, $3.76 f_c$, and $6.52 f_c$, together with negative-frequency reflections across $0\,\text{Hz}$ at $|1 - 2.76|f_c = 1.76 f_c$ and $|1 - 5.52|f_c = 4.52 f_c$ (with phase inversion $\sin(-\omega t) = -\sin(\omega t)$), reproducing metallic bell eigenmodes with two sine evaluators.
\end{corollary}

\section{Amplitude Modulation, LFO, and Deterministic Noise}

\begin{definition}[Modulation and Stochastic Primitives]
\begin{enumerate}[leftmargin=1.5em]
  \item \textbf{Amplitude Modulation / Tremolo (\texttt{AM}, Eq.~19):} $y[n] = x[n]\cdot\big((1-d) + d\sin(2\pi f_{\mathrm{am}} n / f_s)\big)$, $d \in [0, 1]$, generating sidebands at $f \pm f_{\mathrm{am}}$ (used at $32\,\text{Hz}$ in \texttt{shimFrogCroak}).
  \item \textbf{Pitch LFO (\texttt{LFO}):} $\varphi_{\mathrm{mod}}(t) = \varphi(t) + d\sin(2\pi f_{\mathrm{LFO}} t)$ (delayed violin vibrato at $5.8\,\text{Hz}$ and Swarmatron ribbon drift).
  \item \textbf{Deterministic White Noise (\texttt{NSE}):} $\xi[n] \in [-1, 1]$ generated by the 32-bit \texttt{xorshift32} recurrence $s \leftarrow s \oplus (s \ll 13)$; $s \leftarrow s \oplus (s \gg 17)$; $s \leftarrow s \oplus (s \ll 5)$ from a fixed seed $s_0 \ne 0$.
\end{enumerate}
\end{definition}

\begin{proposition}[Bit-Exact Reproducibility of Stochastic Percussion]
Because \texttt{xorshift32} is a pure bijection on $\mathbb{F}_2^{32} \setminus \{0\}$ with period $2^{32}-1$, seeding $s_0$ deterministically at export start guarantees bit-identical noise bursts for \texttt{shimSnare} and \texttt{shimHiHat} across live playback and offline WAV export.
\end{proposition}

\section{Biquad Filters, BIBO Stability, and the Two $Q$ Conventions}

\begin{definition}[Direct Form~I Biquad Filter (Equation~13)]
A \emph{second-order recursive (biquad) filter} is the causal LTI system defined by the difference equation
\begin{equation}
\label{eq:biquad}
y[n] = \frac{b_0}{a_0}x[n] + \frac{b_1}{a_0}x[n-1] + \frac{b_2}{a_0}x[n-2] - \frac{a_1}{a_0}y[n-1] - \frac{a_2}{a_0}y[n-2],
\end{equation}
with $z$-domain transfer function $H(z) = \frac{b_0 + b_1 z^{-1} + b_2 z^{-2}}{a_0 + a_1 z^{-1} + a_2 z^{-2}}$ and Bristow-Johnson Cookbook coefficients (Table~\ref{tab:biquad}) for $\omega_0 = 2\pi f_0 / f_s$, $c = \cos\omega_0$, $\alpha = \sin\omega_0 / (2Q)$, and $A = 10^{G_{\mathrm{dB}}/40}$.
\end{definition}

\begin{table}[htbp]
\centering
\caption{Biquad filter coefficients ($a_1 = -2c$ in all four types; normalized by $a_0$).}
\label{tab:biquad}
\begin{tabular}{@{}llllll@{}}
\toprule
Type & $b_0$ & $b_1$ & $b_2$ & $a_0$ & $a_2$ \\
\midrule
Low-Pass (\texttt{LPF}) & $(1-c)/2$ & $1-c$ & $(1-c)/2$ & $1+\alpha$ & $1-\alpha$ \\
High-Pass (\texttt{HPF}) & $(1+c)/2$ & $-(1+c)$ & $(1+c)/2$ & $1+\alpha$ & $1-\alpha$ \\
Band-Pass (\texttt{BPF}) & $\alpha$ & $0$ & $-\alpha$ & $1+\alpha$ & $1-\alpha$ \\
Peaking EQ (\texttt{PEAK}) & $1+\alpha A$ & $-2c$ & $1-\alpha A$ & $1+\alpha/A$ & $1-\alpha/A$ \\
\bottomrule
\end{tabular}
\end{table}

\begin{theorem}[BIBO Stability of Cookbook Biquads]
\label{thm:biquad-stable}
For every center frequency $0 < f_0 < f_s / 2$ (so $0 < \omega_0 < \pi$), quality factor $Q > 0$, and finite gain $G_{\mathrm{dB}} \in [-24, +24]$, all four biquad filters in Table~\ref{tab:biquad} are strictly Bounded-Input Bounded-Output (BIBO) stable: every pole $p$ of $H(z)$ lies strictly inside the open unit disk $|p| < 1$.
\end{theorem}

\begin{proof}
Let $\tilde{\alpha} = \alpha$ for LP/HP/BP and $\tilde{\alpha} = \alpha/A$ for Peaking EQ. Since $0 < \omega_0 < \pi$, $\sin\omega_0 > 0$, and because $Q > 0$ and $A > 0$, we have $\tilde{\alpha} > 0$. The denominator polynomial is $D(z) = (1+\tilde{\alpha})z^2 - 2c z + (1-\tilde{\alpha})$. By the Jury stability criterion for a real quadratic $a_0 z^2 + a_1 z + a_2$ with $a_0 > 0$, both roots lie strictly inside $|z| < 1$ if and only if:
\begin{enumerate}[leftmargin=1.5em]
  \item $|a_2| < a_0$: since $\tilde{\alpha} > 0$, $-(1+\tilde{\alpha}) < 1 - \tilde{\alpha} < 1 + \tilde{\alpha}$ holds identically;
  \item $D(1) = (1+\tilde{\alpha}) - 2c + (1-\tilde{\alpha}) = 2(1 - \cos\omega_0) > 0$ since $\omega_0 \in (0, \pi)$;
  \item $D(-1) = (1+\tilde{\alpha}) + 2c + (1-\tilde{\alpha}) = 2(1 + \cos\omega_0) > 0$ since $\omega_0 \in (0, \pi)$.
\end{enumerate}
All three Jury conditions hold strictly, proving $|p_{1,2}| < 1$ and hence BIBO stability.
\end{proof}

\begin{proposition}[Equivalence of Linear $Q$ and Decibel $Q_{\mathrm{dB}}$ Conventions]
The \emph{Orchestra} EQ insert computes $\alpha = \sin\omega_0/(2Q)$ with linear $Q$, whereas the legacy synth primitive \texttt{wa\_filter} computes $\alpha = \sin\omega_0/(2\cdot 10^{Q_{\mathrm{dB}}/20})$ for LP/HP. Both filters have identical transfer functions $H(z)$ if and only if $Q_{\mathrm{dB}} = 20\log_{10} Q$ (e.g., $Q = 2.2 \iff Q_{\mathrm{dB}} = 6.848\,\text{dB}$ for \texttt{shimCello}).
\end{proposition}

\section{Non-Linear WaveShaper Saturation and Soft Limiting}

\begin{definition}[WaveShaper and Master Soft Limiter (Equations~14 and~20)]
For drive $k \in [0, 100]$, the \emph{B-Sound WaveShaper} $C_k : [-1, 1] \to \mathbb{R}$ is
\begin{equation}
\label{eq:waveshaper}
C_k(x) = \frac{(3 + k)\,x\cdot (\pi / 9)}{\pi + k\,|x|}, \qquad \frac{\pi}{9} = 20^\circ.
\end{equation}
The \emph{master soft limiter} $\mathrm{lim} : \mathbb{R} \to \mathbb{R}$ with linear threshold $\theta = 0.85$ is
\begin{equation}
\label{eq:limiter}
\mathrm{lim}(x) =
\begin{cases}
x, & |x| \le 0.85, \\
\mathrm{sgn}(x)\!\left(0.85 + \dfrac{|x| - 0.85}{1 + 2.5(|x| - 0.85)}\right), & |x| > 0.85,
\end{cases}
\end{equation}
followed by 16-bit PCM quantization $s = \mathrm{lrintf}\big(\mathrm{clamp}(\mathrm{lim}(x), -1, 1)\times 32767\big)$.
\end{definition}

\begin{theorem}[Symmetry, Monotonicity, and Compression Bounds]
\label{thm:nonlin-bounds}
Both $C_k$ ($k \ge 0$) and $\mathrm{lim}$ are odd, $C^1$-smooth, strictly increasing functions. Specifically:
\begin{enumerate}[leftmargin=1.5em]
  \item $C_k(-x) = -C_k(x)$, so $C_k$ applied to a pure sinusoid generates exclusively odd harmonics ($3f, 5f, \dots$), and $C_k'(x) = \frac{\pi(3+k)(\pi/9)}{(\pi + k|x|)^2} > 0$.
  \item $\mathrm{lim}$ is $C^1$ at $|x| = 0.85$ (left and right derivatives both equal $1$), satisfies $\mathrm{lim}(1.25) = 1.05$, and is strictly bounded asymptotically by $|\mathrm{lim}(x)| < 0.85 + \frac{1}{2.5} = 1.25$ for all $x \in \mathbb{R}$.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) Oddness follows from $|{-x}| = |x|$. For $x > 0$, differentiating $\frac{(3+k)(\pi/9)x}{\pi + kx}$ by the quotient rule gives $\frac{(3+k)(\pi/9)\pi}{(\pi + kx)^2} > 0$, and by oddness the same formula holds for $x < 0$ and at $x = 0$.
(2) For $d = x - 0.85 > 0$, $\frac{d}{dd}\!\left(0.85 + \frac{d}{1+2.5d}\right) = \frac{1}{(1+2.5d)^2} > 0$, which equals $1$ at $d = 0^+$, matching the linear slope $1$ at $x = 0.85^-$ and proving $C^1$ smoothness. Finally, $\frac{d}{1 + 2.5d} = \frac{1}{1/d + 2.5} < \frac{1}{2.5} = 0.40$ for all $d > 0$, so $|\mathrm{lim}(x)| < 0.85 + 0.40 = 1.25$, and at $x = 1.25$ ($d = 0.40$), $\mathrm{lim}(1.25) = 0.85 + 0.40/2.0 = 1.05$.
\end{proof}

\section{Stateful Time-Domain Operators and Schroeder Hall Reverb}

\begin{definition}[Delay, Chorus, Gate, and Schroeder Hall Reverb (Equations~15--18)]
\begin{align}
\text{Tempo Delay (Eq.~16):}\quad & L = \mathrm{round}(k \cdot T_{\mathrm{16th}} \cdot f_s), \quad k \in \{1,\dots,16\}, \\
\text{LFO Chorus (Eq.~17):}\quad & d[n] = d_0 + d_{\mathrm{mod}}\sin(2\pi f_{\mathrm{LFO}} n / f_s), \\
\text{Envelope Gate (Eq.~18):}\quad & e[n] = \max(|x[n]|,\; e[n-1]\cdot \lambda), \quad \lambda = e^{-1/(\tau f_s)}, \\
\text{Hall Reverb (Eq.~15):}\quad & y_{\mathrm{wet}}[n] = \mathrm{Schroeder}_{4\text{C}+4\text{AP}}(x_{\mathrm{send}}[n]) - x_{\mathrm{send}}[n],
\end{align}
where each damped comb branch $c \in \{1,2,3,4\}$ satisfies $b_c[n] = x[n] + g \cdot \mathrm{LP}(b_c[n - D_c])$ with feedback gain $g \in (0, 1)$ ($g = 0.78$ default) and delay length $D_c \in \mathbb{N}_{>0}$ scaled by $f_s / 44100$.
\end{definition}

\begin{theorem}[Stability and Reverberation Time $RT_{60}$ of the Comb Network]
\label{thm:reverb-rt60}
If $|g| < 1$ and the in-loop low-pass damping filter satisfies $|\mathrm{LP}(e^{i\omega})| \le 1$ for all $\omega$, then every comb branch $c$ is BIBO stable and its low-frequency $60\,\text{dB}$ reverberation decay time is
\begin{equation}
RT_{60}^{(c)} = \frac{-3\,D_c / f_s}{\log_{10} g}.
\end{equation}
\end{theorem}

\begin{proof}
At DC ($\omega = 0$, $\mathrm{LP}(1) = 1$), the comb transfer function is $H_c(z) = \frac{z^{-D_c}}{1 - g z^{-D_c}}$, whose $D_c$ poles satisfy $z^{D_c} = g$, so every pole has modulus $|p| = |g|^{1/D_c} < 1$, proving BIBO stability. Since $|\mathrm{LP}(e^{i\omega})| \le 1$, high-frequency loop gain $|g\cdot \mathrm{LP}(e^{i\omega})| \le |g| < 1$ preserves stability across all frequencies. An impulse circulating $m$ times attenuates by $g^m$ in time $t = m D_c / f_s$; solving $20\log_{10}(g^m) = -60$ gives $m = -3 / \log_{10} g$, yielding $RT_{60}^{(c)} = (-3 D_c / f_s)/\log_{10} g$.
\end{proof}
