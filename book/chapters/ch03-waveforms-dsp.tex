\chapter{Analytical Waveforms, Spectral Synthesis, and Digital Signal Processing}
\label{ch:waveforms-dsp}

From the algebraic pitch frequencies $f(n)$ of Chapter~\ref{ch:group-theory}, a Virtual DSP constructs time-domain waveforms $w(\varphi)$, shapes their harmonic spectra through linear time-invariant (LTI) filters and non-linear wave-shapers, and embeds them in an acoustic space via delay networks. This chapter derives all thirteen DSP primitives implemented in B-Sound.

\section{Fourier Series of Canonical Waveforms}

Let $\varphi(t) = 2\pi \int_0^t f(\tau)\,d\tau \pmod{2\pi}$ denote the continuous phase accumulator. Continuous phase integration eliminates click discontinuities during pitch bends and vibrato. The four classical periodic generators in B-Sound are:

\begin{enumerate}[leftmargin=1.5em]
  \item \textbf{Pure Sine (\texttt{SINE}):} $w_{\mathrm{sin}}(\varphi) = \sin(\varphi)$, containing only the fundamental $k = 1$. For sub-bass frequencies ($f < 95\,\text{Hz}$), B-Sound optionally injects $10\%$ second harmonic, $0.90\sin(\varphi) + 0.10\sin(2\varphi)$, to maintain audibility on compact speakers.
  \item \textbf{Triangle (\texttt{TRI}):} $w_{\mathrm{tri}}(\varphi) = \frac{2}{\pi}\arcsin(\sin(\varphi))$, with Fourier expansion containing only odd harmonics decaying quadratically ($-12\,\text{dB/oct}$):
  \begin{equation}
  w_{\mathrm{tri}}(\varphi) = \frac{8}{\pi^2}\sum_{k=0}^{\infty} \frac{(-1)^k}{(2k+1)^2}\sin\!\big((2k+1)\varphi\big).
  \end{equation}
  \item \textbf{Band-Limited Sawtooth (\texttt{SAW}):} An ideal sawtooth decays slowly as $1/k$ ($-6\,\text{dB/oct}$), causing severe aliasing above the Nyquist frequency $f_s / 2$. B-Sound synthesizes an alias-free band-limited partial sum:
  \begin{equation}
  w_{\mathrm{saw}}(\varphi) = 0.62 \sum_{k=1}^{4} \frac{(-1)^{k-1}}{k}\sin(k\varphi),
  \end{equation}
  retaining bright string and brass overtones without foldover distortion.
  \item \textbf{Band-Limited Pulse / Square (\texttt{PLS}):} Rather than a discontinuous $\mathrm{sgn}(\sin\varphi)$ step, B-Sound employs hyperbolic soft-clipping,
  \begin{equation}
  w_{\mathrm{pls}}(\varphi) = 0.78 \tanh(3.4\sin\varphi),
  \end{equation}
  yielding a rich odd-harmonic square timbre with $C^\infty$ smoothness and natural high-frequency roll-off.
\end{enumerate}

\section{Frequency Modulation (FM) and Bessel Sideband Theory}

Two-operator Frequency Modulation (Chowning, 1973) generates complex inharmonic bell and metallic spectra with only two sine evaluators (Equation~12 of \emph{Orchestra}):
\begin{equation}
\label{eq:fm-synth}
y_{\mathrm{FM}}(t) = A_c(t)\sin\!\Big(2\pi f_c t + I(t)\sin(2\pi f_m t)\Big),
\end{equation}
where $f_c$ is the carrier frequency, $f_m = r f_c$ is the modulator frequency, and $I(t)$ is the modulation index.

\begin{theorem}[Jacobi--Anger Spectral Expansion of FM]
For constant modulation index $I$, the FM waveform decomposes into an infinite discrete spectrum at sideband frequencies $f_c + m f_m$ ($m \in \mathbb{Z}$):
\begin{equation}
\sin\!\big(\omega_c t + I\sin(\omega_m t)\big) = \sum_{m=-\infty}^{\infty} J_m(I)\sin\!\big((\omega_c + m\omega_m)t\big),
\end{equation}
where $J_m(I)$ is the Bessel function of the first kind of order $m$, satisfying $J_{-m}(I) = (-1)^m J_m(I)$.
\end{theorem}

\begin{example}[B-Sound \texttt{shimPulseBell}, GM Program 98]
Setting the irrational ratio $r = f_m / f_c = 2.76$ and index $I = 1.8$ produces strong inharmonic partials at $f_c, 3.76 f_c, 6.52 f_c$ alongside reflected negative-frequency sidebands at $|f_c - 2.76 f_c| = 1.76 f_c$ and $|f_c - 5.52 f_c| = 4.52 f_c$, accurately matching the eigenmodes of a struck bronze bell with zero wavetables.
\end{example}

\section{Amplitude Modulation, LFO Vibrato, and Stochastic Noise}

\begin{itemize}[leftmargin=1.5em]
  \item \textbf{Amplitude Modulation / Tremolo (\texttt{AM}, Eq.~19):} Modulates an oscillator or channel bus by $y[n] \leftarrow y[n] \cdot \big((1 - d) + d\sin(2\pi f_{\mathrm{am}} n / f_s)\big)$. At $f_{\mathrm{am}} = 32\,\text{Hz}$, \texttt{shimFrogCroak} synthesizes cu\'ica friction rasp sidebands.
  \item \textbf{Pitch LFO / Vibrato (\texttt{LFO}):} Modulates instantaneous phase via $\varphi_{\mathrm{mod}}(t) = \varphi(t) + d\sin(2\pi f_{\mathrm{LFO}} t)$, modelling delayed violin vibrato ($f_{\mathrm{LFO}} = 5.8\,\text{Hz}$) and Swarmatron ribbon drift.
  \item \textbf{Deterministic Noise (\texttt{NSE}):} White noise $\xi[n] \in [-1, 1]$ is generated by a deterministic 32-bit \texttt{xorshift32} PRNG (or a pre-computed 2-second buffer), guaranteeing bit-identical snare wire and hi-hat transients across offline WAV renders.
\end{itemize}

\section{Biquad Filters and the Two $Q$-Factor Conventions}

Linear spectral shaping in B-Sound relies on the Direct Form~I second-order IIR (biquad) difference equation (Equation~13):
\begin{equation}
\label{eq:biquad}
y[n] = \frac{b_0}{a_0}x[n] + \frac{b_1}{a_0}x[n-1] + \frac{b_2}{a_0}x[n-2] - \frac{a_1}{a_0}y[n-1] - \frac{a_2}{a_0}y[n-2],
\end{equation}
with $z$-domain transfer function $H(z) = (b_0 + b_1 z^{-1} + b_2 z^{-2}) / (a_0 + a_1 z^{-1} + a_2 z^{-2})$. Using Robert Bristow-Johnson's \emph{Audio EQ Cookbook} with $\omega_0 = 2\pi f_0 / f_s$, $c = \cos\omega_0$, $\alpha = \sin\omega_0 / (2Q)$, and $A = 10^{G_{\mathrm{dB}}/40}$, the four filter topologies are:

\begin{table}[htbp]
\centering
\caption{Biquad filter coefficients ($a_1 = -2c$ in all four types; normalized by $a_0$).}
\begin{tabular}{@{}llllll@{}}
\toprule
Type & $b_0$ & $b_1$ & $b_2$ & $a_0$ & $a_2$ \\
\midrule
Low-Pass (\texttt{LPF}) & $(1-c)/2$ & $1-c$ & $(1-c)/2$ & $1+\alpha$ & $1-\alpha$ \\
High-Pass (\texttt{HPF}) & $(1+c)/2$ & $-(1+c)$ & $(1+c)/2$ & $1+\alpha$ & $1-\alpha$ \\
Band-Pass (\texttt{BPF}) & $\alpha$ & $0$ & $-\alpha$ & $1+\alpha$ & $1-\alpha$ \\
Peaking EQ (\texttt{PEAK}) & $1+\alpha A$ & $-2c$ & $1-\alpha A$ & $1+\alpha/A$ & $1-\alpha/A$ \\
\bottomrule
\end{tabular}
\end{table}

\begin{remark}[Critical Implementation Trap: Two $Q$ Conventions]
In B-Sound's C99 runtime, the EQ insert uses linear $Q$ ($\alpha = \sin\omega_0 / (2Q)$), whereas the legacy \texttt{wa\_filter} inside shims accepts $Q$ in decibels for LP and HP ($\alpha = \sin\omega_0 / (2 \cdot 10^{Q_{\mathrm{dB}}/20})$) and linear $Q$ for BP. Furthermore, \texttt{wa\_filter} has no peaking mode; peaking exists exclusively in the EQ insert. Therefore a Web Audio shim parameter $Q = 2.2$ must be translated via $Q_{\mathrm{dB}} = 20\log_{10}(2.2)$ when entering \texttt{wa\_filter}.
\end{remark}

\section{Non-Linear WaveShaper Saturation and Soft Limiting}

To impart analog harmonic warmth and industrial overdrive without digital wrap-around clipping, B-Sound implements a rational odd-symmetric WaveShaper (Equation~14):
\begin{equation}
\label{eq:waveshaper}
C_k(x) = \frac{(3 + k)\,x\cdot (20\pi / 180)}{\pi + k\,|x|}, \qquad x \in [-1, 1], \quad k \in [0, 100].
\end{equation}
At $k = 25$ (the default industrial drive), $C_{25}(x)$ is pre-sampled into a 1024-entry lookup table \texttt{curve[1024]} at boot and updated only on the UI tick, keeping the audio hot path allocation-free.

At the master output stage, the soft limiter (Equation~20) passes signals unchanged for $|x| \le 0.85$ and smoothly compresses excess amplitude $d = |x| - 0.85 > 0$ via:
\begin{equation}
\label{eq:limiter}
\mathrm{lim}(x) = \mathrm{sgn}(x)\!\left(0.85 + \frac{d}{1 + 2.5\,d}\right), \qquad s = \mathrm{lrintf}\big(\mathrm{clamp}(x, -1, 1) \times 32767\big).
\end{equation}
At $x = 1.25$ ($d = 0.40$), $\mathrm{lim}(1.25) = 0.85 + 0.40/2.0 = 1.05$, bounding peak excursions before 16-bit integer quantization.

\section{Time-Domain Delay, Chorus, Gate, and Schroeder Hall Reverb}

Four stateful time-domain processors complete the B-Sound DSP algebra:
\begin{align}
\text{Tempo Delay (Eq.~16):}\quad & L = \mathrm{round}(k \cdot T_{\mathrm{16th}} \cdot f_s), \quad k \in \{1,\dots,16\}, \\
\text{LFO Chorus (Eq.~17):}\quad & d[n] = d_0 + d_{\mathrm{mod}}\sin(2\pi f_{\mathrm{LFO}} n / f_s), \\
\text{Envelope Gate (Eq.~18):}\quad & e[n] = \max(|x[n]|,\; e[n-1]\cdot \lambda), \quad \lambda = e^{-1/(\tau f_s)}, \\
\text{Hall Reverb (Eq.~15):}\quad & y_{\mathrm{wet}}[n] = \mathrm{Schroeder}_{4\text{C}+4\text{AP}}(x_{\mathrm{send}}[n]) - x_{\mathrm{send}}[n].
\end{align}
The reverberator consists of four parallel damped comb filters feeding four series all-pass diffusers per stereo channel, with delay lengths scaled by $f_s / 44100$. Subtracting $x_{\mathrm{send}}$ isolates the pure wet return because the underlying \texttt{wa\_reverb\_process} sums dry and wet internally.
