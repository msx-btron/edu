\section{4D Hopf Fibration and Singularity-Free Stereographic Projection}

\begin{definition}[Hopf Fibration and B-System Nested-Torus Parameterization]
The \emph{Hopf fibration} $\pi : S^3 \to S^2$ decomposes the unit 3-sphere $S^3 = \{(x_0, x_1, x_2, x_3) \in \mathbb{R}^4 : \sum_{j=0}^3 x_j^2 = 1\}$ into linked great circles ($S^1$ fibers). For discrete torus index $t \in \{0,\dots,N_{\mathrm{tori}}-1\}$, fiber index $f \in \{0,\dots,N_{\mathrm{fibers}}-1\}$, and sample $p \in \{0,\dots,N_{\mathrm{pts}}\}$, B-System defines the angular coordinates
\begin{align}
\eta(t) &= \frac{\pi}{7} + (t + 0.6)\frac{\pi}{2.8\,N_{\mathrm{tori}}}, &
\beta(f) &= \frac{2\pi f}{N_{\mathrm{fibers}}} + \omega, &
\xi(p) &= \frac{2\pi p}{N_{\mathrm{pts}}},
\end{align}
with $\omega = 0.08\,\tau + 0.40\,\Phi_c$, $\psi = 0.12\,\tau + \Phi_c$ (flywheel damping $\dot{\Phi}_c \leftarrow 0.968\,\dot{\Phi}_c$), and embeds them into $\mathbb{R}^4$ via
\begin{align}
\label{eq:hopf-s3}
x_0 &= \cos\eta\cos(\xi + \beta/2 + \psi), & x_1 &= \cos\eta\sin(\xi + \beta/2 + \psi), \\
x_2 &= \sin\eta\cos(\xi - \beta/2 + 0.8\psi), & x_3 &= \sin\eta\sin(\xi - \beta/2 + 0.8\psi).
\end{align}
The \emph{regularized stereographic projection} $\sigma_\varepsilon : S^3 \to \mathbb{R}^3$ with pole offset $\varepsilon = 0.08$ is defined by
\begin{equation}
\label{eq:hopf-proj}
\sigma_\varepsilon(x_0, x_1, x_2, x_3) = (X, Y, Z) = \frac{(x_0, x_1, x_2)}{1.08 - x_3}.
\end{equation}
\end{definition}

\begin{theorem}[Unit-Sphere Invariance and Bounded Stereographic Radius]
\label{thm:hopf-bound}
For all $(\eta, \beta, \xi, \omega, \psi) \in \mathbb{R}^5$, the point $\mathbf{x} = (x_0, x_1, x_2, x_3)$ of \eqref{eq:hopf-s3} lies identically on $S^3$, and its regularized stereographic image \eqref{eq:hopf-proj} is globally smooth ($C^\infty$) and strictly bounded in Euclidean norm by
\begin{equation}
\|(X, Y, Z)\|_2 \;=\; \frac{\sqrt{1 - x_3^2}}{1.08 - x_3} \;\le\; \frac{1}{\sqrt{1.08^2 - 1}} \;\approx\; 2.4515 \;<\; 12.5.
\end{equation}
\end{theorem}

\begin{proof}
Computing the squared 4D norm directly gives
\[
x_0^2 + x_1^2 + x_2^2 + x_3^2 = \cos^2\eta\big(\cos^2(\cdot) + \sin^2(\cdot)\big) + \sin^2\eta\big(\cos^2(\cdot) + \sin^2(\cdot)\big) = \cos^2\eta + \sin^2\eta = 1,
\]
so $\mathbf{x} \in S^3$ and $x_3 \in [-1, 1]$. Hence $1.08 - x_3 \in [0.08, 2.08]$ never vanishes, making $\sigma_\varepsilon$ globally $C^\infty$. Since $x_0^2 + x_1^2 + x_2^2 = 1 - x_3^2$, maximize $g(x_3) = \frac{1 - x_3^2}{(1.08 - x_3)^2}$ on $[-1, 1]$. Differentiating yields $g'(x_3) = \frac{2(1 - 1.08 x_3)}{(1.08 - x_3)^3} = 0$ at $x_3^* = 1/1.08 \in (-1, 1)$, where
\[
g(x_3^*) = \frac{1 - 1/1.08^2}{(1.08 - 1/1.08)^2} = \frac{1.08^2 - 1}{(1.08^2 - 1)^2} = \frac{1}{1.08^2 - 1} = \frac{1}{0.1664} \approx 6.0096.
\]
Taking square roots gives $\|(X, Y, Z)\|_2 \le \sqrt{6.0096} \approx 2.4515$, proving the entire 4D fiber bundle fits strictly inside the OpenGL frustum ($z_{\mathrm{cam}} = 9.2$, near $= 1.0$, far $= 100.0$) with zero pole singularities.
\end{proof}

\section{Low-Poly Geometry and 24-DOF Critically Damped Kinematics}

\begin{definition}[Radial Chamfer Operator and 24-DOF State Vector]
The mascot torso is generated from the 26 boundary vertices $V \in \{-1.05, 0, +1.05\}^3 \setminus \{(0,0,0)\}$ via the \emph{radial bevel clamp} at $R_{\mathrm{bevel}} = 1.65$:
\begin{equation}
\mathcal{C}_{R}(V) = V \cdot \min\!\left(1,\; \frac{R_{\mathrm{bevel}}}{\|V\|_2}\right),
\end{equation}
yielding $< 180$ triangles enclosing a counter-rotating regular octahedron ($R_{\mathrm{gem}} = 0.78$) beneath a 24-slice toroidal rainbow halo ($R = 0.95, r = 0.05$). The complete mascot pose is a continuous state vector $\mathbf{x}(t) \in \mathbb{R}^{24}$ encoding root position/orientation, torso offset/rotation, volumetric scale $(S_x, S_y, S_z)$, hand/foot joints, and mouth morph weights.
\end{definition}

\begin{definition}[Critically Damped Relaxation Dynamics]
Each state coordinate $x_i(t)$ approaches its target pose $x_i^*(t)$ under the critically damped ($\zeta = 1.0$) linear ODE
\begin{equation}
\label{eq:crit-damp}
\ddot{x}_i + 2\omega_i\dot{x}_i + \omega_i^2\big(x_i(t) - x_i^*(t)\big) = 0,
\end{equation}
discretized at frame step $\Delta t > 0$ ($16.67\,\text{ms}$ at $60\,\text{FPS}$) by the exponential relaxation operator
\begin{equation}
\label{eq:exp-relax}
\mathcal{T}_{\Delta t}(x_i) = x_i^*(t) + \big(x_i(t) - x_i^*(t)\big)e^{-k_i \Delta t},
\end{equation}
with stiffness constants $(k_{\mathrm{pos}}, k_{\mathrm{rot}}, k_{\mathrm{scale}}, k_{\mathrm{hand}}, k_{\mathrm{foot}}, k_{\mathrm{mouth}}) = (18, 16, 22, 15, 20, 24)\,\text{s}^{-1}$.
\end{definition}

\begin{theorem}[Zero-Overshoot Contraction and Frame-Rate Independence]
\label{thm:kin-contract}
For every constant target $x_i^*$ and stiffness $k_i > 0$:
\begin{enumerate}[leftmargin=1.5em]
  \item The continuous ODE \eqref{eq:crit-damp} has a double real characteristic root $\lambda = -\omega_i < 0$, so its error $e_i(t) = x_i(t) - x_i^* = (c_1 + c_2 t)e^{-\omega_i t}$ converges to $0$ with at most one sign change and zero periodic oscillation.
  \item The discrete update \eqref{eq:exp-relax} is a strict Banach contraction with Lipschitz constant $\rho = e^{-k_i \Delta t} \in (0, 1)$, and satisfies the exact semigroup property $\mathcal{T}_{\Delta t_1} \circ \mathcal{T}_{\Delta t_2} = \mathcal{T}_{\Delta t_1 + \Delta t_2}$, guaranteeing frame-rate-independent trajectories across arbitrary render cadences.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) The characteristic polynomial of \eqref{eq:crit-damp} is $\lambda^2 + 2\omega_i\lambda + \omega_i^2 = (\lambda + \omega_i)^2 = 0$, giving the repeated negative real root $\lambda = -\omega_i$; hence no sinusoidal $\cos/\sin$ factors arise.
(2) For any $u, v \in \mathbb{R}$, $|\mathcal{T}_{\Delta t}(u) - \mathcal{T}_{\Delta t}(v)| = e^{-k_i \Delta t}|u - v| = \rho |u - v|$ with $\rho < 1$ since $k_i, \Delta t > 0$. Composing two steps gives
\[
\mathcal{T}_{\Delta t_1}(\mathcal{T}_{\Delta t_2}(x)) = x^* + \big((x - x^*)e^{-k_i \Delta t_2}\big)e^{-k_i \Delta t_1} = x^* + (x - x^*)e^{-k_i(\Delta t_1 + \Delta t_2)} = \mathcal{T}_{\Delta t_1 + \Delta t_2}(x),
\]
proving exact invariance under frame-rate subdivision.
\end{proof}

\section{The Seven-State Automaton and Interactive Beat Quantization}

The mascot finite state machine $\mathcal{K}_7$ comprises seven states phase-locked to the shared audio clock \texttt{wa\_time() - s\_tl\_base} ($\approx 0.7\,\text{ms}$ granularity):
\begin{enumerate}[leftmargin=1.5em]
  \item \textbf{IDLE:} Isovolumetric breathing $S_y = 1 + 0.04\sin(3.2t)$, $S_{xz} = 1 - 0.02\sin(3.2t)$ paired with \texttt{shimPulseBell}.
  \item \textbf{WADDLE:} Stride phase $\Phi_w(t) = \int 7.5\,dt$; zero-crossings of $\sin\Phi_w$ fire \texttt{shimWaddle} on alternating root/fifth bass notes.
  \item \textbf{FLOAT:} $3.35\,\text{s}$ four-phase spline levitation paired with \texttt{shimFloat} Rhodes chords and touchdown \texttt{shimBoing}.
  \item \textbf{DANCE:} $4.10\,\text{s}$ pirouette choreography firing 16th-note \texttt{shimDance} star arpeggios every $90^\circ$ rotation.
  \item \textbf{SQUASH / BLESSING:} Isovolumetric compression ($S_y = 0.60, S_{xz} = 1.30$, $S_y S_{xz}^2 \approx 1.014$) to elongation ($S_y = 1.45, S_{xz} = 0.82$), firing \texttt{shimPulseSquash} and a spherical radiant wave $r(t) = 0.5 + (1 - (1 - t/T)^{2.8})(R_{\max} - 0.5)$.
  \item \textbf{SING:} Wide O-mouth ($w_{\mathrm{omouth}} = 1.25$) with $5\times$ internal gem acceleration on fortissimo climaxes.
  \item \textbf{POKE:} Viscoelastic damped oscillation $\mathrm{squish}(t) = 0.36\,e^{-4.5t}\cos(22t)$ decaying in $0.55\,\text{s}$.
\end{enumerate}

\begin{definition}[Forward 16th-Note Input Quantizer]
Given a bar origin $t_{\mathrm{bar}}$, 16th-note period $T_{\mathrm{16th}} = 15/\mathrm{BPM}$, and asynchronous user event at $t_{\mathrm{input}} \ge t_{\mathrm{bar}}$, the \emph{interactive quantizer} $Q_{16}$ schedules the audio-visual response at
\begin{equation}
\label{eq:quant-16}
Q_{16}(t_{\mathrm{input}}) = t_{\mathrm{bar}} + \left\lceil \frac{t_{\mathrm{input}} - t_{\mathrm{bar}}}{T_{\mathrm{16th}}} \right\rceil T_{\mathrm{16th}}.
\end{equation}
\end{definition}

\begin{theorem}[Causality and Bounded Metric Quantization Delay]
\label{thm:quant-delay}
For every $t_{\mathrm{input}} \ge t_{\mathrm{bar}}$, the scheduled instant $Q_{16}(t_{\mathrm{input}})$ is strictly causal, lies exactly on the 16th-note metric lattice $t_{\mathrm{bar}} + \mathbb{Z}_{\ge 0} T_{\mathrm{16th}}$, and incurs latency strictly bounded by
\begin{equation}
0 \;\le\; Q_{16}(t_{\mathrm{input}}) - t_{\mathrm{input}} \;<\; T_{\mathrm{16th}} = \frac{15}{\mathrm{BPM}}\,\text{s}.
\end{equation}
\end{theorem}

\begin{proof}
Let $u = (t_{\mathrm{input}} - t_{\mathrm{bar}})/T_{\mathrm{16th}} \ge 0$. By definition of the ceiling function, $u \le \lceil u \rceil < u + 1$ with $\lceil u \rceil \in \mathbb{Z}_{\ge 0}$. Multiplying by $T_{\mathrm{16th}} > 0$ and subtracting $(t_{\mathrm{input}} - t_{\mathrm{bar}})$ yields $0 \le Q_{16}(t_{\mathrm{input}}) - t_{\mathrm{input}} < T_{\mathrm{16th}}$. At $168\,\text{BPM}$ (Track~1), the worst-case quantization wait is $< 89.3\,\text{ms}$.
\end{proof}
