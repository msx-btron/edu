\section{The Functional Pitch-Curve Synthesizer Model}

\begin{definition}[Functional Pitch-Curve Synthesizer]
A \emph{functional pitch-curve synthesizer} is a five-tuple $\mathcal{F} = (X, Y, \mathcal{E}, \Phi^{-1}, \mathcal{M})$, where:
\begin{enumerate}[leftmargin=1.5em]
  \item $X = [-8, +8]$ is the normalized horizontal time-cursor domain scanned at demo speed $v_x > 0$ ($\Delta x = v_x / f_s$ per sample);
  \item $Y = [-3, +3]$ is the visible vertical octave domain anchored at middle $\mathrm{C}_4$ ($F_0 = 261.6256\,\text{Hz}$), with active synthesis acceptance band $(-4.5, +4.5)$;
  \item $\mathcal{E} : X \times \{0, \dots, P-1\} \to (\mathbb{R} \cup \{\mathrm{NaN}\})^{N_v}$ evaluates $N_v \le 10$ parallel voice curves $y_v(x)$ for part index $p$, where $\mathrm{NaN}$ denotes a musical rest;
  \item $\Phi^{-1}(y) = F_0 \cdot 2^y$ maps each finite curve ordinate to instantaneous frequency in Hz (Equation~\ref{eq:log2-iso});
  \item $\mathcal{M} : \{0,\dots,N_v-1\} \to \Sigma_{\mathrm{DSP}}$ assigns each voice a synthesis primitive label from the 15-symbol DSP alphabet $\Sigma_{\mathrm{DSP}}$ (augmented by optional per-voice gain $g_v$ and echo operator $\mathcal{E}_v$).
\end{enumerate}
\end{definition}

\begin{theorem}[Click-Free Continuity of Functional Rest Transitions and Scrubbing]
\label{thm:func-synth-click}
Let each voice envelope $E_v[n] \in [0, 1]$ evolve under the one-pole exponential smoother
\begin{equation}
E_v[n] = E_v[n-1] + \big(\mathbf{1}_{\mathrm{active}}(v, n) - E_v[n-1]\big)\cdot\Big(1 - e^{-1/(0.0028\,f_s)}\Big),
\end{equation}
with continuous phase integration $\varphi_v[n] = \varphi_v[n-1] + 2\pi \Phi^{-1}(y_v(x[n]))/f_s \pmod{2\pi}$. Then every transition between a finite curve $y_v(x)$ and a rest $\mathrm{NaN}$, as well as arbitrary interactive pointer scrubbing $x(t)$, produces a bounded-slew output free of step-click discontinuities, settling within $5\tau = 14\,\text{ms}$.
\end{theorem}

\begin{proof}
Because $1 - e^{-1/(0.0028 f_s)} \in (0, 1)$, $E_v[n]$ is a convex combination of $E_v[n-1]$ and $\{0, 1\}$ with time constant $\tau = 2.8\,\text{ms}$. Across any $\mathrm{NaN}$ boundary, the sample-to-sample amplitude jump satisfies $|\Delta E_v| \le 1 - e^{-1/(0.0028 f_s)} \approx 0.0074$ at $f_s = 48000\,\text{Hz}$, suppressing broadband click energy while preserving crisp percussive attacks. By Theorem~\ref{thm:waveforms}, continuous phase integration preserves $C^0$ waveform continuity during live frequency glides and scrubbing.
\end{proof}

\section{The 15 Pro-Audio DSP Labels, Live Echo Heads, and Unified Telemetry Console}

To bridge mathematical equations with professional audio engineering practice, every demo in the Functional Synthesizer renders a live \textbf{DSP Voice Strip} and \textbf{Broadcast Metering Strip} integrated directly into the bottom console panel (\texttt{\#formula-screen}) so the upper $\mathrm{C}_7$ pitch space remains unobstructed.

\begin{definition}[The 15-Symbol DSP Primitive Alphabet $\Sigma_{\mathrm{DSP}}$ and Independent Echo Heads]
Every voice $v$ displays a color-coded IEC~60617 / Eurorack geometric vector marker drawn from the closed 15-element alphabet $\Sigma_{\mathrm{DSP}}$ (Table~\ref{tab:dsp-labels}), whose chip opacity toggles live between $1.0$ (sounding) and $0.34$ (rest). Whenever a voice carries a per-voice echo $\mathcal{E}_v$ (Definition~\ref{def:echo}), two synchronized visual elements verify its timing on the square musical scale grid:
\begin{enumerate}[leftmargin=1.5em]
  \item \textbf{Shifted Ghost Echo Trails:} For repeats $k \in \{1, 2, 3, 4\}$, dashed copies of curve $y_v(x)$ are plotted shifted right by $\Delta x_k = k \cdot v_x \cdot \tau_v$ with geometric opacity $\alpha_k \propto w_v g_v^{k-1}$.
  \item \textbf{Separate Live \texttt{ECHO} Heads:} Each echoing voice spawns an independent colored \texttt{ECHO <ms>} HUD chip and concentric hollow playhead ring that illuminate strictly when dry voice $v$ has finished and at least one delayed tap $y_v(x - \Delta x_k)$ is sounding—allowing a human listener to verify visually where dry \texttt{PNO} ends and \texttt{ECHO} takes over.
\end{enumerate}
\end{definition}

\begin{definition}[Broadcast Metering Telemetry]
Per audio buffer of length $N = 1024$ frames, the telemetry engine computes:
\begin{align}
\mathrm{PK}_{\mathrm{dBFS}} &= \max\!\big(-60,\; 20\log_{10}\max(P_L, P_R)\big), \\
\mathrm{RMS}_{\mathrm{dBFS}} &= \max\!\Big(-60,\; 20\log_{10}\sqrt{\tfrac{1}{2}(\mathrm{RMS}_L^2 + \mathrm{RMS}_R^2)}\Big), \\
\mathrm{LIM} &= \mathbf{1}\big[\max(|x_L|, |x_R|) > 0.85\big], \qquad
\mathrm{bal} = \frac{\mathrm{RMS}_R - \mathrm{RMS}_L}{\mathrm{RMS}_R + \mathrm{RMS}_L} \in [-1, 1],
\end{align}
with ballistic peak decay $0.84$ per block and stereo balance indicator rendered alongside dual L/R micro-bargraphs calibrated from $-48\,\text{dBFS}$ to $0\,\text{dBFS}$ (highlighting hot peaks above $85\%$ in warning red).
\end{definition}

\subsection*{The 18-Part \emph{Analytical Etudes} (Demo 16)}

Demo~16 serves as the pedagogical bridge between raw analytical functions and classical/electronic composition across four phases:
\begin{enumerate}[leftmargin=1.5em]
  \item \textbf{Phase I (Parts 1--4 — From Chaos to Pulse):} Opens with the raw unquantized nested-sine carrier $y = 2.8\sin(5\sin(4\sin(3\sin(2\sin x))))$ (\texttt{Broken Radio I}), then tames it into consonant shortwave beacons ($\mathrm{Am9}$), 16th-note chiptune arpeggios ($\mathrm{Em}$), and Steve Reich variable-subdivision pentatonic grooves.
  \item \textbf{Phase II (Parts 5--8 — Minimalist Ostinatos):} Harmonic overtone bloom ($\mathrm{A}_1 \to \mathrm{E}_5$), damped exponential piano pulses (\emph{Time}), 3/4 organ/celesta cascades (\emph{Cornfield Chase}), and WaveShaper sub-brass swells (\emph{Dune}).
  \item \textbf{Phase III (Parts 9--12 — Piecewise Orchestration):} Music-box fifths (\emph{S.T.A.Y.}), logistic-sigmoid vocal portamento $y_a + (y_b - y_a)/(1 + e^{-14\tau})$ (\emph{Gladiator}), 3:4 organ polyrhythms (\emph{No Time For Caution}), and a D-minor string canon (\emph{Sangreal}).
  \item \textbf{Phase IV (Parts 13--18 — Climax, Physics, and Mosaic):} Driving ostinato chase (\emph{Mombasa}), relativistic pendulum (\emph{Mountains}), heroic D-major fanfare (\emph{Flight}), 8-voice cathedral climax (\emph{Grand Symphony}), Aphex Twin's geometric gravity cascade $n = \lfloor \ln(1 - \tau/T)/\ln(0.78)\rfloor$ quantized onto $\mathrm{F}\sharp\mathrm{m9}/\mathrm{Dmaj9}$ (\emph{Bucephalus Bouncing Ball}, Part~17), and a braided macro-wave mosaic of short hyperbolic $1/(t+0.30)$ and exponential $e^{\pm kt}$ pieces floating over warm constant $\mathrm{Am9}\cdot\mathrm{F}\cdot\mathrm{C}\cdot\mathrm{G}$ harmonic lines (Part~18).
\end{enumerate}

\begin{proposition}[Hands-Free Curriculum Traversal vs.\ Local Part Cycling]
In \texttt{[AUTO]} mode, when the cursor reaches $x \ge +8$ on the final part of demo $d$, both the Web Audio sample clock and the visual fallback clock advance automatically to $(d+1) \bmod 26$, part $0$, resetting voice envelopes, per-voice echo rings, and hall buffers so a listener can audit all 148 parts continuously without intervention. By contrast, manual \texttt{ArrowDown} (\texttt{selectPart}) computes $(p + 1) \bmod P_d$, cycling strictly within the current demo—matching the interaction contract of \texttt{Physics.html}.
\end{proposition}
