\chapter{The Procedural Sound Shim Architecture}
\label{ch:shims}

Having established the computational envelope of Chapter~\ref{ch:intro}, the algebraic pitch groups of Chapter~\ref{ch:group-theory}, and the thirteen DSP primitives of Chapter~\ref{ch:waveforms-dsp}, we now formalize the central synthesis abstraction of B-Sound: the \textbf{Sound Shim}, full mathematical specifications for all 23 canonical shims, the immutable 32-program General MIDI registry, and the statically bounded 48-slot voice allocator.

\section{Formal Definition and Finite-Energy Properties of Sound Shims}

\begin{definition}[Procedural Sound Shim (Equations~10--11)]
A \emph{Sound Shim} is a stateless synthesis operator $S : \mathbb{R}_{\ge 0} \times \mathbb{R}^d \to L^2(\mathbb{R})$ that, given a trigger instant $t_0 \ge 0$ and a finite parameter vector $\mathbf{p} = (f, v, d, \dots) \in \mathbb{R}^d$, constructs a causal acoustic signal $y(t) = S(t_0, \mathbf{p})(t)$ by finite partial superposition:
\begin{equation}
\label{eq:shim-def}
y(t) = \mathbf{1}_{t \ge t_0} \sum_{k=1}^{K} \Big( A_k(t - t_0)\cdot W_k\!\big(\varphi_k(t - t_0)\big) \Big) * h_k(t),
\end{equation}
where $K \in \mathbb{N}_{>0}$ is a fixed compile-time partial count, $W_k \in \{w_{\mathrm{sin}}, w_{\mathrm{tri}}, w_{\mathrm{saw}}, w_{\mathrm{pls}}, y_{\mathrm{FM}}, \xi\}$ is a bounded primitive ($|W_k| \le 1$), $h_k \in L^1(\mathbb{R})$ is the impulse response of a BIBO-stable biquad filter (or $\delta_0$ if unfiltered), and $A_k$ is an exponential ADSR envelope with peak $V_k > 0$, attack $t_{\mathrm{att}, k} \ge 0$, and decay constant $\tau_k > 0$:
\begin{equation}
\label{eq:shim-env}
A_k(s) =
\begin{cases}
V_k \cdot \dfrac{s}{t_{\mathrm{att}, k}}, & 0 \le s < t_{\mathrm{att}, k}, \\[6pt]
V_k \cdot \exp\!\left(-\dfrac{s - t_{\mathrm{att}, k}}{\tau_k}\right), & s \ge t_{\mathrm{att}, k}.
\end{cases}
\end{equation}
\end{definition}

Throughout the shim specifications below, let $s = t - t_0 \in [0, d]$ denote local elapsed time, let $\mathrm{expRamp}(a, b, s, T) = a\,(b/a)^{s/T}$ denote an exponential trajectory from $a$ to $b$ over duration $T$, and let $\mathrm{linRamp}(a, b, s, T) = a + (b - a)(s/T)$ denote a linear ramp.

\begin{theorem}[$L^2$-Integrability and Peak Boundedness of Sound Shims]
\label{thm:shim-l2}
Every B-Sound procedural shim $S(t_0, \mathbf{p})$ with finite partial count $K$, bounded primitives $|W_k| \le 1$, decay constants $\tau_k > 0$, and BIBO-stable filters $\|h_k\|_{L^1} < \infty$ satisfies $y \in L^1(\mathbb{R}) \cap L^2(\mathbb{R}) \cap L^\infty(\mathbb{R})$, with explicit peak bound
\begin{equation}
\|y\|_{L^\infty} \;\le\; \sum_{k=1}^{K} V_k \, \|h_k\|_{L^1} \;<\; \infty.
\end{equation}
\end{theorem}

\begin{proof}
For each partial $k$, let $u_k(s) = A_k(s) W_k(\varphi_k(s))$ for $s \ge 0$. Since $|W_k| \le 1$ and $0 \le A_k(s) \le V_k$, we have $\|u_k\|_{L^\infty} \le V_k$, $\|u_k\|_{L^1} \le V_k(t_{\mathrm{att},k}/2 + \tau_k) < \infty$, and $\|u_k\|_{L^2}^2 \le V_k^2(t_{\mathrm{att},k}/3 + \tau_k/2) < \infty$. By Young's convolution inequality, for $p \in \{1, 2, \infty\}$ and BIBO-stable $h_k \in L^1$ (Theorem~\ref{thm:biquad-stable}), $\|u_k * h_k\|_{L^p} \le \|u_k\|_{L^p}\|h_k\|_{L^1} < \infty$. Summing over $k = 1,\dots,K$ yields the claim.
\end{proof}

\section{Family I: Kinematic Mascot Shims (1A--5 and Boing)}
\label{sec:fam-kin}

\begin{definition}[Shim 1A — \texttt{shimPulseSquash}$(t_0, v_{\mathrm{mul}}, \rho)$]
A 6-partial additive singing C-major chord ($\mathrm{C}_4, \mathrm{C}_5, \mathrm{E}_5, \mathrm{G}_5, \mathrm{C}_6, \mathrm{E}_6$) with upward $3\%$ scoop:
\begin{align}
y(s) &= \sum_{k=1}^{6} A_k(s)\sin\!\big(\varphi_k(s)\big), \\
f_k(s) &=
\begin{cases}
\mathrm{expRamp}(0.97\rho F_k,\, \rho F_k,\, s,\, 0.10), & s < 0.10, \\
\rho F_k, & s \ge 0.10,
\end{cases}
\end{align}
with partials $(F_k, V_k, d_k)$ given by $(261.63, 0.055, 2.2)$, $(523.25, 0.070, 2.4)$, $(659.25, 0.048, 2.0)$, $(783.99, 0.052, 2.1)$, $(1046.50, 0.040, 1.7)$, and $(1318.51, 0.025, 1.4)$. Each partial ramps linearly from $10^{-4}$ to $v_{\mathrm{mul}} V_k$ in $0.09\,\text{s}$ and decays exponentially to $10^{-4}$ at $s = d_k$.
\end{definition}

\begin{definition}[Shim 1A-Alt — \texttt{shimFrogCroak}$(t_0, \rho, v_{\mathrm{mul}}, \mathrm{dbl})$]
A call-and-response cu\'ica / frog rasp emitting Stroke~1 at $\Delta t_1 = 0\,\text{s}$ ($f_b = 125\,\text{Hz}, d = 0.19\,\text{s}, \alpha = 1.0$) and, if $\mathrm{dbl} = 1$, Stroke~2 at $\Delta t_2 = 0.25\,\text{s}$ ($f_b = 155\,\text{Hz}, d = 0.095\,\text{s}, \alpha = 0.85$). Each stroke sums a sawtooth throat oscillator ($f_1 = \rho f_b$) and triangle mouth overtone ($f_2 = 2.45\rho f_b$) through a bandpass cavity filter $H_{\mathrm{BP}}(f_c = 2.2\rho f_b, Q = 3.8)$:
\begin{align}
f_1(s) &=
\begin{cases}
\mathrm{expRamp}(0.88 f_1,\, 1.25 f_1,\, s,\, 0.75d), & s < 0.75d, \\
\mathrm{linRamp}(1.25 f_1,\, 0.95 f_1,\, s - 0.75d,\, 0.25d), & 0.75d \le s \le d,
\end{cases} \\
f_2(s) &= \mathrm{expRamp}\!\big(0.90 f_2,\, 1.20 f_2,\, \min(s, 0.75d),\, 0.75d\big).
\end{align}
The output gain is modulated by a $32.26\,\text{Hz}$ ($\Delta p = 0.031\,\text{s}$) triangular AM rasp train of $N_p = \lfloor d / 0.031 \rfloor$ pulses with peak $P_p = 0.24\,v_{\mathrm{mul}}\alpha(1 - 0.07p)$, rising $0.2 P_p \to P_p$ in $9\,\text{ms}$ and falling to $0.15 P_p$ at $25\,\text{ms}$, followed by a $40\,\text{ms}$ exponential release.
\end{definition}

\begin{definition}[Shim 1B — \texttt{shimPulseBell}$(t_0, v_{\mathrm{mul}}, \rho)$]
B-Sound specifies two dual realizations of \texttt{shimPulseBell}:
\begin{enumerate}[leftmargin=1.5em]
  \item \textbf{Web Audio Additive Swell:} An 8-partial sine overtone series with $(F_k, V_k, d_k)$ equal to $(130.81, 0.065, 3.5)$, $(196.00, 0.050, 3.2)$, $(261.63, 0.070, 3.4)$, $(392.00, 0.055, 3.0)$, $(523.25, 0.060, 2.8)$, $(659.25, 0.040, 2.5)$, $(783.99, 0.035, 2.3)$, and $(1046.50, 0.025, 1.9)$. Each partial scoops $0.985\rho F_k \to \rho F_k$ exponentially in $0.14\,\text{s}$ with a $0.12\,\text{s}$ linear attack.
  \item \textbf{C99 Orchestra 2-Op FM (GM Program~98):}
  \[
  y(s) = v_{\mathrm{mul}} e^{-s/1.8}\sin\!\big(2\pi f_c s + 1.8\sin(2\pi \cdot 2.76 f_c s)\big)
  \]
  (Theorem~\ref{thm:jacobi-anger}).
\end{enumerate}
\end{definition}

\begin{definition}[Shim 2 — \texttt{shimWaddle}$(t_0, f_0, v_{\mathrm{mul}}, d)$]
A bipedal step and upright walking bass ($f_0 = 430\,\text{Hz}, d = 0.08\,\text{s}$ default):
\begin{align}
y(s) &= \mathrm{expRamp}(0.12 v_{\mathrm{mul}}, 0.001, s, d)\cdot\big(w_{\mathrm{tri}}(\varphi(s)) * h_{\mathrm{LP}, 180}\big), \\
f(s) &= \mathrm{expRamp}(f_0, 0.82 f_0, s, d),
\end{align}
where the $180\,\text{Hz}$ low-pass body filter is active in the C99 bass voice (\texttt{shimWaddleD}, GM Program~32).
\end{definition}

\begin{definition}[Shim 3 — \texttt{shimFloat}$(t_0, f_0, v_{\mathrm{mul}}, \mathcal{C}, \delta, d)$]
A staggered Rhodes / glass-harp chord. For default $\mathcal{C} = \emptyset$ and $\rho = f_0/523.25$, four sine voices enter at $t_k \in \{0, 0.06, 0.12, 0.18\}\,\text{s}$ with $(F_k, V_k, d_k)$ given by $(523.25\rho, 0.12, 0.55)$, $(659.25\rho, 0.10, 0.50)$, $(783.99\rho, 0.11, 0.65)$, and $(1046.50\rho, 0.08, 0.75)$; each scoops $0.985 F_k \to F_k$ in $40\,\text{ms}$, attacks in $30\,\text{ms}$, and decays exponentially to $10^{-4}$.
\end{definition}

\begin{definition}[Shim 4 — \texttt{shimDance}$(t_0, \rho, v_{\mathrm{mul}}, \mathcal{N})$]
A 5-note crystalline D-major cascade ($\mathrm{D}_5, \mathrm{F}\sharp_5, \mathrm{A}_5, \mathrm{D}_6, \mathrm{F}\sharp_6$) with default tuples $(t_k, F_k, V_k, d_k)$:
\begin{align*}
&(0,\, 587.33\rho,\, 0.16,\, 0.20), \quad (0.085,\, 739.99\rho,\, 0.15,\, 0.22), \quad (0.170,\, 880.00\rho,\, 0.16,\, 0.25), \\
&(0.255,\, 1174.66\rho,\, 0.18,\, 0.32), \quad (0.370,\, 1479.98\rho,\, 0.12,\, 0.45),
\end{align*}
using pure sine oscillators with $20\,\text{ms}$ linear attack and exponential release to $10^{-4}$.
\end{definition}

\begin{definition}[Shim 5 — \texttt{shimDing}$(t_0, f_0, v_{\mathrm{mul}}, d)$ and \texttt{sfxLilCubeBoing}$(t_0)$]
\texttt{shimDing} ($f_0 = 1760\,\text{Hz}, d = 0.70\,\text{s}$) is a pure sine chime with $12\,\text{ms}$ linear attack to $0.15 v_{\mathrm{mul}}$ and exponential decay to $10^{-4}$ at $s = d$. The auxiliary touchdown spring \texttt{sfxLilCubeBoing} drives a triangle oscillator over $d = 0.28\,\text{s}$ from $523.25 \xrightarrow{0.14\,\text{s}} 1174.66 \xrightarrow{0.14\,\text{s}} 783.99\,\text{Hz}$ with exponential amplitude $0.20 \to 0.005$.
\end{definition}

\section{Family II: Percussion DSP Shims (6--8, 12--13, 23)}
\label{sec:fam-perc}

\begin{definition}[Shim 6 — \texttt{shimKick}$(t_0, f_0, v_{\mathrm{mul}}, d)$]
An acoustic/retro kick drum ($f_0 = 65\,\text{Hz}, d = 0.16\,\text{s}$):
\begin{equation}
y(s) = \mathrm{expRamp}(0.24 v_{\mathrm{mul}}, 0.001, s, d)\cdot \sin\!\big(\varphi(s)\big), \qquad
f(s) = \mathrm{expRamp}(2.2 f_0,\, 0.75 f_0,\, s,\, d).
\end{equation}
\end{definition}

\begin{definition}[Shim 7 — \texttt{shimSnare}$(t_0, v_{\mathrm{mul}}, \mathrm{rim})$]
A dual-branch shell + snare-wire synthesizer ($\mathrm{rim} = 1$ when MIDI velocity $v_m \ge 110$):
\begin{equation}
y(s) = A_{\mathrm{sh}}(s)\,w_{\mathrm{tri}}(\varphi_{\mathrm{sh}}(s)) + A_{\mathrm{wr}}(s)\big(\xi * h_{\mathrm{BP}}(f_{\mathrm{bp}}, Q=2.0)\big)(s),
\end{equation}
where $f_{\mathrm{sh}}(s) = \mathrm{expRamp}(f_b, 0.65 f_b, s, 0.12)$ with $f_b = 220\,\text{Hz}$ (rimshot) or $175\,\text{Hz}$ (normal), $A_{\mathrm{sh}}: 0.18 v_{\mathrm{mul}} \to 0.001$ in $0.12\,\text{s}$; and wire branch $f_{\mathrm{bp}} = 3200\,\text{Hz}$ ($d_{\mathrm{wr}} = 0.09\,\text{s}$) for rimshot or $2400\,\text{Hz}$ ($d_{\mathrm{wr}} = 0.15\,\text{s}$) for normal, $A_{\mathrm{wr}}: 0.16 v_{\mathrm{mul}} \to 0.001$.
\end{definition}

\begin{definition}[Shim 8 — \texttt{shimHiHat}$(t_0, \mathrm{open}, v_{\mathrm{mul}})$]
With $d = 0.28\,\text{s}$ ($\mathrm{open} = 1$, MIDI $v_m \ge 100$) or $d = 0.055\,\text{s}$ (closed):
\begin{align}
y(s) &= \mathrm{expRamp}(0.08 v_{\mathrm{mul}}, 5\!\times\!10^{-4}, s, d)(\xi * h_{\mathrm{HP},7500})(s) \nonumber\\
     &\quad + \mathrm{expRamp}(0.04 v_{\mathrm{mul}}, 5\!\times\!10^{-4}, s, d)\sin(2\pi f_m s),
\end{align}
with metallic overtone ping $f_m = 6200\,\text{Hz}$ (open) or $8400\,\text{Hz}$ (closed).
\end{definition}

\begin{definition}[Shim 12 — \texttt{shimIndustrialKick}$(t_0, v_{\mathrm{mul}}, d)$]
A fixed-pitch industrial sub-punch ($d = 0.22\,\text{s}$) summing a sine body ($160 \xrightarrow{\exp} 42\,\text{Hz}$, $0.32 v_{\mathrm{mul}} \to 0.001$ over $d$) with a saturated click transient: triangle $380 \xrightarrow{\exp} 55\,\text{Hz}$ ($50\,\text{ms}$) $\to$ $h_{\mathrm{LP}, 1800}$ $\to$ WaveShaper $C_{25}$ (Eq.~\ref{eq:waveshaper}), gain $0.24 v_{\mathrm{mul}} \to 0.001$ in $55\,\text{ms}$.
\end{definition}

\begin{definition}[Shim 13 — \texttt{shimIndustrialSnare}$(t_0, v_{\mathrm{mul}}, \mathrm{clank})$]
Sums an overdriven noise burst $\xi \to h_{\mathrm{BP}}(f_{\mathrm{bp}}, Q = 2.2) \to C_{30}$ ($f_{\mathrm{bp}} = 3600\,\text{Hz}, d = 0.08\,\text{s}$ for clank; $2600\,\text{Hz}, d = 0.17\,\text{s}$ for backbeat; gain $0.24 v_{\mathrm{mul}} \to 0.001$) with a dissonant metallic trash-can pair $w_{\mathrm{pls}}(2\pi\cdot 220 s) + w_{\mathrm{tri}}(2\pi\cdot 352 s)$, each decaying $0.12 v_{\mathrm{mul}} \to 0.001$ over $0.11\,\text{s}$.
\end{definition}

\begin{definition}[Shim 23 — \texttt{shimTimpani}$(t_0, f_0, v_{\mathrm{mul}}, d)$]
A 3-branch concert kettle drum ($f_0 = 82.41\,\text{Hz}, d = 1.2\,\text{s}$):
\begin{enumerate}[leftmargin=1.5em]
  \item \textbf{Membrane Head:} Sine sweeping $1.55 f_0 \xrightarrow{\exp} f_0$ in $50\,\text{ms}$, gain $0.35 v_{\mathrm{mul}} \to 0.001$ over $d$.
  \item \textbf{Copper Kettle Mode $(1,1)$:} Constant sine at $1.48 f_0$, gain $0.14 v_{\mathrm{mul}} \to 0.001$ over $0.4d$.
  \item \textbf{Felt Mallet Impact:} Noise $\xi * h_{\mathrm{LP}, 260}$, gain $0.18 v_{\mathrm{mul}} \to 0.001$ in $45\,\text{ms}$.
\end{enumerate}
\end{definition}

\section{Family III: Acoustic and Orchestral Shims (9--11, 18--22)}
\label{sec:fam-orch}

\begin{definition}[Shim 9 — \texttt{shimGuitar}$(t_0, \mathcal{N}, v_{\mathrm{mul}})$]
Each note $(f_0, d, v, f_{\mathrm{slide}})$ drives a triangle oscillator (optional exponential pitch slide to $f_{\mathrm{slide}}$ over $0.7d$) through a time-varying low-pass filter sweeping $f_c(s) = \mathrm{expRamp}(2800, 1400, s, d)\,\text{Hz}$, with $15\,\text{ms}$ linear thumb-pick attack to $v\cdot v_{\mathrm{mul}}$ and exponential release to $10^{-4}$.
\end{definition}

\begin{definition}[Shim 10 — \texttt{shimGrandPiano}$(t_0, f_0, v_{\mathrm{mul}}, d, \mathrm{stacc})$]
A 3-partial stiff-string model with natural inharmonicity ratios $r_h \in \{1.0, 2.002, 3.006\}$, waveforms $(w_{\mathrm{sin}}, w_{\mathrm{tri}}, w_{\mathrm{sin}})$, gains $V_h \in \{0.16, 0.08, 0.035\}$, effective duration $d_0 = \min(d, 0.18)$ if $\mathrm{stacc}=1$ else $d$, and upper-partial damping $d_h = d_0(1 - 0.25 h)$ ($h \in \{0,1,2\}$). Each partial has an $8\,\text{ms}$ wooden-hammer linear attack and exponential decay to $10^{-4}$.
\end{definition}

\begin{definition}[Shim 11 — \texttt{shimPedalBass}$(t_0, f_0, v_{\mathrm{mul}}, d)$]
A sforzando sub-bass ($f_0 = 55\,\text{Hz}$, $d = 2.2\,\text{s}$): triangle oscillator with subtle $2\%$ string-tension sag $f(s) = \mathrm{expRamp}(f_0, 0.98 f_0, s, d)$ and exponential envelope from $0.20 v_{\mathrm{mul}}$ to $5\times 10^{-4}$.
\end{definition}

\begin{definition}[Shim 18 — \texttt{shimSaxFlute}$(t_0, \mathcal{N}, v_{\mathrm{mul}})$]
Each note sums a triangle body branch (optional $0.75d$ slide) through $h_{\mathrm{LP}}(2400\,\text{Hz}, Q=2.0)$ ($25\,\text{ms}$ attack to $v\cdot v_{\mathrm{mul}}$) with a parallel breath-turbulence branch $\xi * h_{\mathrm{BP}}(1.8 f_0, Q = 5.0)$ decaying exponentially $0.025 v_{\mathrm{mul}} \to 5\times 10^{-4}$ over $d$.
\end{definition}

\begin{definition}[Shims 19 \& 20 — Helper Piano Mechanics]
\texttt{shimPianoHammer}$(t_0, v_{\mathrm{mul}})$ generates a $35\,\text{ms}$ felt thump via triangle $120 \xrightarrow{\exp} 40\,\text{Hz}$ (gain $0.12 v_{\mathrm{mul}} \to 5\times 10^{-4}$). \texttt{shimPedalSwoosh}$(t_0, v_{\mathrm{mul}}, d=0.40)$ models damper lift via $\xi * h_{\mathrm{BP}}(f_c(s), Q=3.5)$ with $f_c(s) = \mathrm{expRamp}(450, 280, s, d)\,\text{Hz}$ and $80\,\text{ms}$ attack to $0.05 v_{\mathrm{mul}}$. Neither helper is mapped directly in \texttt{PROG[32]}.
\end{definition}

\begin{definition}[Shim 21 — \texttt{shimViolin}$(t_0, \mathcal{N}, v_{\mathrm{mul}}, \mathrm{solo})$]
A five-stage bowed string model:
\begin{enumerate}[leftmargin=1.5em]
  \item \textbf{Rosin Transient:} $\xi * h_{\mathrm{BP}}(3200\,\text{Hz}, Q=3.0)$, $8\,\text{ms}$ attack to $0.045 v_{\mathrm{mul}}$, decay to $5\times 10^{-4}$ at $40\,\text{ms}$.
  \item \textbf{Dual Dispersive Saws:} $w_{\mathrm{saw}}(\varphi_1) + w_{\mathrm{saw}}(\varphi_2)$ at $(f_0, 1.002 f_0)$ ($+3.46\,\text{cents}$ detune; optional $0.85d$ slide).
  \item \textbf{Delayed Vibrato LFO:} $5.8\,\text{Hz}$ sine modulating both frequencies after delay $\min(0.08, 0.3d)\,\text{s}$, ramping to depth $0.012 f_0$ (solo) or $0.008 f_0$ (section) by $\min(0.24, 0.7d)\,\text{s}$.
  \item \textbf{Body Formants:} Series cascade
  \begin{align*}
  h_{\mathrm{PEAK}}(520\,\text{Hz}, Q=2.2, +4\,\text{dB})
  &\;\to\; h_{\mathrm{PEAK}}(2800\,\text{Hz}, Q=1.8, +3\,\text{dB}) \\
  &\;\to\; h_{\mathrm{LP}}(6200 / 5000\,\text{Hz}).
  \end{align*}
  (In C99 \emph{Orchestra}, where \texttt{wa\_filter} lacks peaking, \texttt{shimViolin} uses a 3-node noise/2-saw/LP reduction and delegates body resonance to the track EQ insert.)
  \item \textbf{Bowed Envelope:} Linear attack $\min(0.045, 0.25d)\,\text{s}$ to $v$, sustain $0.90v$ at $0.75d$, exponential release to $5\times 10^{-4}$.
\end{enumerate}
\end{definition}

\begin{definition}[Shim 22 — \texttt{shimCello}$(t_0, \mathcal{N}, v_{\mathrm{mul}}, \mathrm{pizz})$]
Sums unison sawtooth and triangle oscillators into $h_{\mathrm{LP}}(f_c, Q=2.2)$ with $f_c = 1200\,\text{Hz}$ (pizzicato) or $520\,\text{Hz}$ (arco). Pizzicato uses an $8\,\text{ms}$ pluck to $1.3v$ with exponential release; arco uses a $\min(0.08, 0.3d)\,\text{s}$ bow swell to $v$, sustain $0.95v$ at $0.7d$, and exponential release to $5\times 10^{-4}$.
\end{definition}

\section{Family IV: Industrial and Swarm Shims (14--17)}

\begin{definition}[Shim 14 — \texttt{shimIndustrialBass}$(t_0, f_0, d, v_{\mathrm{mul}}, f_{\mathrm{slide}})$]

\section{The Immutable 32-Program Registry (\texttt{PROG[32]})}

\begin{definition}[Immutable Program Registry]
The \emph{B-Sound Program Registry} is a partial function
\begin{equation}
\mathcal{R}_{\mathrm{GM}} : \{0, 1, \dots, 127\} \rightharpoonup \mathcal{S} \times \{\mathrm{tr}, \mathrm{bs}\} \times \mathbb{N}_{>0} \times \mathbb{Z} \times \{0, 1\}
\end{equation}
defined on a fixed domain $\mathrm{dom}(\mathcal{R}_{\mathrm{GM}})$ of cardinality $|\mathrm{dom}(\mathcal{R}_{\mathrm{GM}})| = 32$, assigning each registered General MIDI program $g$ a tuple $(S_g, \mathrm{clef}_g, p_{\max,g}, \Delta_{\mathrm{prog},g}, \mathrm{shot}_g)$. For any $g \notin \mathrm{dom}(\mathcal{R}_{\mathrm{GM}})$, the registry returns $\bot$, increments \texttt{refuse\_program}, and preserves the track's prior state.
\end{definition}

\begin{proposition}[Semantic Soundness of Partial GM Registration]
Because $\mathcal{R}_{\mathrm{GM}}$ rejects all $96$ unmapped General MIDI program numbers rather than substituting a silent default, every accepted session file establishes an exact semantic bijection between its stored program bytes and the measured WCET-budgeted shims of the running engine. Furthermore, omitting a hardcoded MIDI Channel~10 drum map makes channel assignment orthogonal to timbre, permitting Channel~10 to carry French Horn (Program~60) in the philharmonic template.
\end{proposition}

\section{48-Slot Voice Allocation and Loudness-Decay Stealing}

\begin{definition}[Static 48-Slot Pool and Loudness-Decay Proxy (Equation~9)]
The DAW voice pool is the static array $\mathcal{V}_{48} = \{s_0, s_1, \dots, s_{47}\}$, where each slot $s_i$ stores either $\emptyset$ (free) or an active tuple $(\mathrm{trk}_i, \mathrm{key}_i, g_i, t_{0,i}, t_{\mathrm{exp},i})$. At sample position $n_{\mathrm{pos}} \ge t_{0,i}$, the \emph{loudness-decay proxy} with time constant $\tau = 1\,\text{s}$ is
\begin{equation}
\label{eq:voice-proxy}
\mathrm{proxy}(s_i, n_{\mathrm{pos}}) = g_i \cdot \exp\!\left(-\frac{n_{\mathrm{pos}} - t_{0,i}}{f_s}\right).
\end{equation}
\end{definition}

\begin{definition}[Three-Tier Deterministic Allocator $\alpha_{\mathrm{alloc}}$]
Given an incoming note on track $r$ with per-track ceiling $p_{\max}(r)$, the allocator $\alpha_{\mathrm{alloc}}(r, n_{\mathrm{pos}})$ selects a target slot index $i^* \in \{0,\dots,47\}$ by:
\begin{enumerate}[leftmargin=1.5em]
  \item \textbf{Per-Track Ceiling:} If $|\{i : \mathrm{trk}_i = r\}| \ge p_{\max}(r)$, set $i^* = \arg\min_{i:\,\mathrm{trk}_i = r} \mathrm{proxy}(s_i, n_{\mathrm{pos}})$.
  \item \textbf{Free Scan:} Else if $\exists i$ with $s_i = \emptyset$, set $i^* = \min\{i : s_i = \emptyset\}$.
  \item \textbf{Global Steal:} Else set $i^* = \arg\min_{0 \le i < 48} \mathrm{proxy}(s_i, n_{\mathrm{pos}})$ and increment \texttt{steal\_voice} (unless \texttt{s\_export == 1}).
\end{enumerate}
\end{definition}

\begin{theorem}[$O(1)$ Termination, Pool Conservation, and Track Isolation]
\label{thm:voice-alloc}
For any sequence of note-on events across arbitrary tracks, $\alpha_{\mathrm{alloc}}$ satisfies:
\begin{enumerate}[leftmargin=1.5em]
  \item \textbf{Bounded $O(1)$ Execution:} $\alpha_{\mathrm{alloc}}$ terminates in at most $48$ slot inspections with zero heap allocations.
  \item \textbf{Strict Pool Conservation:} The number of active DAW slots never exceeds $48$, and if the invoked shim fails to land a synth node, the slot is immediately returned to $\emptyset$ and counted in \texttt{shim\_landed}.
  \item \textbf{Track-First Steal Isolation:} No track exceeding its own polyphony ceiling $p_{\max}(r)$ can ever steal a slot belonging to another track $r' \ne r$.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) Each of the three branches scans at most the fixed 48-element array $\mathcal{V}_{48}$, taking strictly $O(48) = O(1)$ time.
(2) Because every allocation either overwrites an existing slot $s_{i^*}$ or occupies an empty slot in $\mathcal{V}_{48}$, $|\{i : s_i \ne \emptyset\}| \le 48$ is a loop invariant; returning unlanded slots prevents leaked occupancy.
(3) Branch~1 is evaluated strictly before Branches~2 and~3; hence whenever track $r$ holds $p_{\max}(r)$ voices, $i^*$ is chosen solely from $\{i : \mathrm{trk}_i = r\}$, leaving every track $r' \ne r$ untouched.
\end{proof}

\begin{definition}[One-Shot Expiry Semantics]
A program $g$ is \emph{one-shot} if $\mathrm{shot}_g = 1$ (the nine rows: \texttt{47} Timpani $1.2\,\text{s}$, \texttt{80} Pulse $2.4\,\text{s}$, \texttt{102} Dance $0.85\,\text{s}$, \texttt{103} Zap $0.12\,\text{s}$, \texttt{112} Chime $0.70\,\text{s}$, \texttt{116} Kick $0.16\,\text{s}$, \texttt{117} Snare $0.15\,\text{s}$, \texttt{118} Hat $0.055/0.28\,\text{s}$, \texttt{119} Ind.\ Kick $0.22\,\text{s}$). One-shot slots ignore MIDI Note-Off messages and are released exclusively when $n_{\mathrm{pos}} \ge t_{\mathrm{exp},i}$ by \texttt{slots\_expire()}.
\end{definition}

\begin{proposition}[Note-Off Invariance of One-Shot Percussion]
Under one-shot semantics, percussive timbre duration is an invariant of the procedural instrument rather than the UI edit grid, preventing short piano-roll notes from truncating natural drum or cymbal tails while still guaranteeing deterministic slot reclamation within $t_{\mathrm{exp},i} - t_{0,i} \le 2.4\,\text{s}$.
\end{proposition}
